cols_shine <- c("#A8E6CF", "#52B788", "#2D7A4F", "#1B4332", "#081C15")

normalizar_nombre <- function(x) {
    x |>
        str_to_lower() |>
        stri_trans_general("Latin-ASCII") |>
        str_squish()
}

# Categorías ocupacionales de la hoja 1.C, únicamente para Rural total.
categorias <- read_excel(
    "Anexo1_OK.xlsx",
    sheet = "1.C",
    col_names = FALSE,
    .name_repair = "minimal"
) |>
    setNames(c(
        "partido", "area", "obrero_publico", "obrero_privado", "patron",
        "cuenta_propia", "familiar_con_sueldo", "familiar_sin_sueldo", "total"
    )) |>
    fill(partido) |>
    filter(area == "Rural total", partido != "Total") |>
    mutate(
        across(obrero_publico:total, ~ suppressWarnings(as.numeric(as.character(.)))),
        partido_id = normalizar_nombre(partido)
    ) |>
    select(-area)

# Distribución por sexo de la hoja 1.D, para las mismas unidades geográficas.
sexo <- read_excel(
    "Anexo1_OK.xlsx",
    sheet = "1.D",
    col_names = FALSE,
    .name_repair = "minimal"
) |>
    setNames(c("partido", "area", "varon", "mujer", "total_sexo")) |>
    fill(partido) |>
    filter(area == "Rural total", partido != "Total") |>
    mutate(
        across(varon:total_sexo, ~ suppressWarnings(as.numeric(as.character(.)))),
        partido_id = normalizar_nombre(partido)
    ) |>
    select(-area)

datos_mapa <- categorias |>
    left_join(sexo |> select(partido_id, varon, mujer, total_sexo), by = "partido_id") |>
    mutate(
        total = coalesce(total, total_sexo),
        pct_varon = varon / total * 100,
        pct_mujer = mujer / total * 100
    )

# Límites oficiales de partidos bonaerenses provistos por geoAr.
pba <- get_geo("BUENOS AIRES", level = "departamento")
codigos <- show_arg_codes(nivel = "departamentos")

mapa_pba <- pba |>
    left_join(
        codigos |> select(coddepto_censo, nomdepto_censo, codprov_censo),
        by = "coddepto_censo"
    ) |>
    filter(codprov_censo.y == "06") |>
    mutate(partido_id = normalizar_nombre(nomdepto_censo)) |>
    left_join(datos_mapa, by = "partido_id")

formatear_numero <- function(x) {
    format(x, big.mark = ".", decimal.mark = ",", scientific = FALSE, trim = TRUE)
}

crear_popup <- function(fila) {
    categorias_popup <- c(
        "Obrero/empleado sector público" = fila$obrero_publico,
        "Obrero/empleado sector privado" = fila$obrero_privado,
        "Patrón" = fila$patron,
        "Trabajador por cuenta propia" = fila$cuenta_propia,
        "Trabajador familiar con sueldo" = fila$familiar_con_sueldo,
        "Trabajador familiar sin sueldo" = fila$familiar_sin_sueldo
    )

    filas_categoria <- paste0(
        "<tr><td>", names(categorias_popup), "</td><td><b>",
        formatear_numero(unname(categorias_popup)), "</b></td></tr>",
        collapse = ""
    )

    HTML(paste0(
        "<div style='min-width:280px'>",
        "<h4 style='margin:0 0 8px'>", htmlEscape(fila$nomdepto_censo), "</h4>",
        "<p style='margin:4px 0'><b>Población activa rural total:</b> ",
        formatear_numero(fila$total), "</p>",
        "<p style='margin:4px 0'><b>Varones:</b> ", formatear_numero(fila$varon),
        " (", round(fila$pct_varon, 1), "%)<br>",
        "<b>Mujeres:</b> ", formatear_numero(fila$mujer),
        " (", round(fila$pct_mujer, 1), "%)</p>",
        "<table style='width:100%;border-collapse:collapse'>",
        "<tr><th style='text-align:left;border-bottom:1px solid #ccc'>Categoría ocupacional</th>",
        "<th style='text-align:right;border-bottom:1px solid #ccc'>Personas</th></tr>",
        filas_categoria,
        "</table></div>"
    ))
}

mapa_pba$popup <- vapply(
    seq_len(nrow(mapa_pba)),
    function(i) crear_popup(mapa_pba[i, ]),
    character(1)
)

paleta <- colorNumeric(

    domain = mapa_pba$total,
    na.color = "#D9D9D9"
